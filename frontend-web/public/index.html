// Slices massive files (270MB+ / 1M+ rows) on disk so the dataset loads in <2 seconds with zero browser lag async function uploadUniversalCsv(event) { const file = event.target.files[0]; if (!file) return; const toast = document.getElementById("etlToast"); const toastText = document.getElementById("etlToastText"); const progressBar = document.getElementById("etlProgressBar"); const sizeMB = (file.size / (1024 * 1024)).toFixed(1); toast.classList.remove("hidden"); progressBar.style.width = "20%"; toastText.innerText = `Reading ${file.name} (${sizeMB} MB)...`; try { // Read only a 4 MB slice from disk (~25,000 complete rows) so it finishes in < 1.5 seconds! const SLICE_BYTES = 4 * 1024 * 1024; const blobSlice = file.slice(0, SLICE_BYTES); let chunkText = await blobSlice.text(); // Trim any partial last line if the file was larger than 4 MB if (file.size > SLICE_BYTES) { const lastNewline = chunkText.lastIndexOf("\n"); if (lastNewline > 0) chunkText = chunkText.slice(0, lastNewline); } progressBar.style.width = "60%"; toastText.innerText = `Processing ${file.name} (${sizeMB} MB) — Generating analytics...`; // Send via same-origin Express gateway (/api/etl-chunk) so port 8000 CORS never blocks it const res = await fetch("/api/etl-chunk", { method: "POST", headers: { "Content-Type": "application/json", "x-api-key": activeApiKey || "key-sample-sales" }, body: JSON.stringify({ filename: file.name, csvChunkText: chunkText }) }); const data = await res.json(); if (res.ok) { progressBar.style.width = "100%"; toastText.innerText = `Loaded ${data.store_name} (${data.load_clean_rows_saved.toLocaleString()} verified rows indexed).`; selectedProducts = []; await refreshDatasetSidebar(data.api_key); await loadStoreData(0); } else { toastText.innerText = `Upload error: ${data.detail || "Failed"}`; } } catch (err) { toastText.innerText = `Upload error: ${err.message}`; } }
